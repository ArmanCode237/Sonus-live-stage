cmake_minimum_required(VERSION 3.16)
project(NordVirtualStage VERSION 28.2.9 LANGUAGES CXX)

# Real-time audio must never ship unoptimized: single-config generators
# (Ninja, MinGW/Unix Makefiles) default to no flags when no type is given.
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
    message(STATUS "Live Stage: CMAKE_BUILD_TYPE not set, defaulting to Release")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)

# Evitar colisión del target "uninstall" entre RtAudio y RtMidi y activar ASIO/WASAPI
set(RTAUDIO_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
# Match the application default /MD runtime when using MSVC (avoid LNK2038).
set(RTAUDIO_STATIC_MSVCRT OFF CACHE BOOL "" FORCE)
set(RTAUDIO_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(RTAUDIO_TARGETNAME_UNINSTALL "uninstall_rtaudio" CACHE STRING "" FORCE)
if(WIN32)
    set(RTAUDIO_API_WASAPI ON CACHE BOOL "" FORCE)
    option(STAGE_ENABLE_ASIO "Build native ASIO output with RtAudio on Windows" ON)
    set(RTAUDIO_API_ASIO ${STAGE_ENABLE_ASIO} CACHE BOOL "" FORCE)
    if(STAGE_ENABLE_ASIO)
        message(STATUS "Live Stage: ASIO + WASAPI enabled. Verify RtAudio/Steinberg license for distribution.")
    else()
        message(WARNING "Live Stage: ASIO DISABLED. Native UMC1820 ASIO driver WILL NOT be listed; only WASAPI is available.")
    endif()
endif()

# MT-32: Munt's libmt32emu (LGPL-2.1+, third_party/mt32emu/COPYING.LESSER.txt), static,
# C++ API only. It needs the user's own MT-32 control + PCM ROM dumps at run time
# (Roland's ROMs are copyrighted and are not part of this project).
set(libmt32emu_SHARED OFF CACHE BOOL "" FORCE)
set(libmt32emu_C_INTERFACE OFF CACHE BOOL "" FORCE)
set(libmt32emu_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(libmt32emu_PKGCONFIG_INSTALL_PREFIX "" CACHE STRING "" FORCE)
add_subdirectory(third_party/mt32emu EXCLUDE_FROM_ALL)

    third_party/msfa/lfo.cc third_party/msfa/pitchenv.cc third_party/msfa/porta.cpp third_party/msfa/sin.cc)
target_link_libraries(stage_msfa PUBLIC mt32emu)
link_libraries(stage_msfa mt32emu)

    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}/webui
    COMMENT "Building the React console into ui/app"
    VERBATIM)
endif()
