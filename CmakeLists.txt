# React control surface (webui/ → ui/app). The built files are committed, so a
# plain C++ build needs no Node.js; `cmake --build build --target webui`
# rebuilds them after editing webui/src.
find_program(STAGE_NPM_EXECUTABLE npm)
if(STAGE_NPM_EXECUTABLE)
  add_custom_target(webui
    COMMAND ${STAGE_NPM_EXECUTABLE} run build
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}/webui
    COMMENT "Building the React console into ui/app"
    VERBATIM)
endif()

# Distribution resources. macOS apps have a read-only Contents/Resources directory;
# never write programs, caches or session files there (use ~/Library/Application Support).
if(APPLE)
  set(STAGE_MAC_APP "Applications/Sonus Live Stage.app")
  set(STAGE_MAC_RES "${STAGE_MAC_APP}/Contents/Resources")
  install(TARGETS SonusLiveStage BUNDLE DESTINATION Applications COMPONENT app)
  install(DIRECTORY ui/app/ DESTINATION "${STAGE_MAC_RES}/ui/app" COMPONENT app)
  install(DIRECTORY presets/ DESTINATION "${STAGE_MAC_RES}/presets" COMPONENT app)
  install(FILES programs.json hardware_config.json DESTINATION "${STAGE_MAC_RES}" COMPONENT app)
  install(FILES config/links.json DESTINATION "${STAGE_MAC_RES}/config" COMPONENT app)
  install(FILES LICENSE docs/GUIA-DE-USO.md docs/THIRD_PARTY_MACOS.md
          DESTINATION "${STAGE_MAC_RES}/docs" COMPONENT app)
  if(EXISTS "${CMAKE_SOURCE_DIR}/NOTICE.md")
    install(FILES NOTICE.md DESTINATION "${STAGE_MAC_RES}/docs" COMPONENT app)
  endif()
  if(EXISTS "${CMAKE_SOURCE_DIR}/roms/mt32/LEEME.txt")
    install(FILES roms/mt32/LEEME.txt DESTINATION "${STAGE_MAC_RES}/roms/mt32" COMPONENT app)
  endif()
else()
  install(TARGETS SonusLiveStage RUNTIME DESTINATION . COMPONENT app)
  install(DIRECTORY ui/app/ DESTINATION ui/app COMPONENT app)
  install(DIRECTORY presets/ DESTINATION presets COMPONENT app)
  install(FILES programs.json hardware_config.json DESTINATION . COMPONENT app)
  install(FILES roms/mt32/LEEME.txt DESTINATION roms/mt32 OPTIONAL COMPONENT app)
  install(FILES docs/AUDITORIA-LIVE-ESTUDIO.md DESTINATION docs OPTIONAL COMPONENT app)
  install(FILES LICENSE DESTINATION . COMPONENT app)
  if(EXISTS "${CMAKE_SOURCE_DIR}/NOTICE.md")
    install(FILES NOTICE.md DESTINATION . COMPONENT app)
  endif()
  install(FILES config/links.json DESTINATION config COMPONENT app)
  install(FILES docs/GUIA-DE-USO.md DESTINATION docs COMPONENT app)
endif()
# Package only the "app" component of this build tree (build dir;project;component;subdir).
set(CPACK_INSTALL_CMAKE_PROJECTS "${CMAKE_BINARY_DIR};SonusLiveStage;app;/")
set(CPACK_PACKAGE_NAME "Sonus Live Stage")
set(CPACK_PACKAGE_VENDOR "Sonus")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_INSTALL_DIRECTORY "Sonus Live Stage")
if(WIN32)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_COMPONENT app)   # MSVC runtime DLLs (MSVC builds only)
  include(InstallRequiredSystemLibraries)
  # ZIP (portable) always; the NSIS installer only when makensis is installed.
  find_program(STAGE_MAKENSIS makensis PATHS "C:/Program Files (x86)/NSIS" "C:/Program Files/NSIS")
  if(STAGE_MAKENSIS)
    set(CPACK_GENERATOR "NSIS;ZIP")
  else()
    set(CPACK_GENERATOR "ZIP")
  endif()
  set(CPACK_NSIS_DISPLAY_NAME "Live Stage")
  set(CPACK_NSIS_PACKAGE_NAME "Sonus Live Stage ${PROJECT_VERSION}")
  set(CPACK_NSIS_INSTALLED_ICON_NAME "SonusLiveStage.exe")
  # Start-menu shortcut "Live Stage" -> SonusLiveStage.exe (NSIS installer).
  set(CPACK_PACKAGE_EXECUTABLES "SonusLiveStage" "Live Stage")
  set(CPACK_PACKAGE_FILE_NAME "SonusLiveStage-${PROJECT_VERSION}-win64")
elseif(APPLE)
  set(CPACK_GENERATOR "ZIP") # scripts/build_macos.sh creates the actual .pkg
else()
  set(CPACK_GENERATOR "TGZ")
endif()
include(CPack)
